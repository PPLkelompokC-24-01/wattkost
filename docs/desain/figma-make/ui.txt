type IconName = "bolt" | "home" | "plug" | "calendar" | "receipt" | "chart" | "target" | "trophy" | "file" | "building" | "bell" | "search" | "logout" | "leaf"

const iconFor = (label = ""): IconName => {
  if (label.includes("Beranda")) return "home"
  if (label.includes("Alat") || label.includes("Katalog")) return "plug"
  if (label.includes("Harian")) return "calendar"
  if (label.includes("Token")) return "receipt"
  if (
    label.includes("Dashboard") ||
    label.includes("Pemakaian") ||
    label.includes("Biaya")
  )
    return "chart"
  if (label.includes("Target") || label.includes("Tips")) return "target"
  if (label.includes("Tantangan") || label.includes("Poin")) return "trophy"
  if (label.includes("Laporan")) return "file"
  if (label.includes("Kost")) return "building"
  if (label.includes("Karbon") || label.includes("Emisi")) return "leaf"
  return "bolt"
}

export function PageTitle({
  title,
  sub,
  children,
}: {
  title: string
  sub?: string
  children?: ReactNode
}) {
  if (!sub && !children) return null
  return (
    <div className="flex items-center justify-between gap-4 mb-6">
      <div>
        {sub && <p className="text-sm text-secondary">{sub}</p>}
      </div>
      {children && <div className="flex items-center gap-3">{children}</div>}
    </div>
  )
}

export function Btn({
  children,
  onClick,
  variant = "primary",
  small,
}: {
  children: ReactNode
  onClick?: () => void
  variant?: "primary" | "secondary" | "link" | "hero" | "danger"
  small?: boolean
}) {
  const base = `${
    small ? "h-8 px-3 text-xs" : "h-11 px-5 text-sm"
  } rounded-full font-semibold inline-flex items-center justify-center gap-2 whitespace-nowrap transition-colors focus:outline-none focus:ring-4 focus:ring-accent-soft`
  const v =
    variant === "primary"
      ? "bg-aqua text-abyss hover:bg-aqua-hover"
      : variant === "hero"
        ? "bg-aqua text-abyss hover:bg-lime"
        : variant === "danger"
          ? "bg-transparent border border-coral/30 text-coral hover:bg-coral-soft"
        : variant === "link"
          ? "text-aqua hover:text-lime px-0 h-auto focus:ring-0"
          : "bg-transparent border border-line-strong text-white hover:bg-white/6 hover:border-aqua"
  return (
    <button onClick={onClick} className={`${base} ${v}`}>
      {children}
    </button>
  )
}

const statTone = (label: string) =>
  label.includes("Sisa") ||
  label.includes("Total kWh") ||
  label.includes("Estimasi Biaya")
    ? "bg-gradient-to-br from-deep to-surface text-white border-line-strong"
    : label.includes("Prediksi") || label.includes("Perbandingan")
      ? "bg-surface border-line"
      : label.includes("Pemakaian") ||
          label.includes("Konsumsi") ||
          label.includes("Emisi")
        ? "bg-surface border-line"
        : "bg-surface border-line"

export function Stat({
  label,
  value,
  note,
  onClick,
  big,
}: {
  label: string
  value: string
  note?: string
  onClick?: () => void
  big?: boolean
}) {
  const tone = statTone(label)
  const dark = true
  return (
    <div
      onClick={onClick}
      className={`min-h-36 border rounded-3xl p-6 relative overflow-hidden ${tone} ${
        onClick
          ? "cursor-pointer transition-transform hover:-translate-y-0.5"
          : ""
      }`}
    >
      <div
        className={`flex items-center gap-3 text-xs font-semibold ${
          dark ? "text-white/75" : "text-mute"
        }`}
      >
        <span
          className={`size-9 rounded-xl grid place-items-center ${
            dark ? "bg-aqua-soft text-aqua" : "bg-aqua-soft text-aqua"
          }`}
        >
          <Icon name={iconFor(label)} />
        </span>
        {label}
      </div>
      <div
        className={`${
          big ? "text-[42px]" : "text-[28px]"
        } font-meter font-medium mt-4 leading-none`}
      >
        {value}
      </div>
      {note && (
        <div className={`text-xs mt-2 ${dark ? "text-white/70" : "text-mute"}`}>
          {note}
        </div>
      )}
      {onClick && (
        <div
          className={`text-xs mt-3 font-bold ${
            dark ? "text-aqua" : "text-aqua"
          }`}
        >
          Lihat detail →
        </div>
      )}
    </div>
  )
}

export function Field({
  label,
  placeholder,
  value,
  suffix,
  type,
}: {
  label: string
  placeholder?: string
  value?: string
  suffix?: string
  type?: "select" | "date" | "area"
}) {
  return (
    <label className="flex flex-col gap-2">
      {label && <span className="text-sm font-semibold">{label}</span>}
      {type === "area" ? (
        <div className="h-24 rounded-[14px] border border-line bg-white/4 px-4 py-3 text-sm text-mute focus-within:border-aqua">
          {value ?? placeholder}
        </div>
      ) : (
        <div className="h-12 rounded-[14px] border border-line bg-white/4 px-4 flex items-center justify-between gap-3 text-sm hover:border-line-strong focus-within:border-aqua focus-within:ring-4 focus-within:ring-aqua-soft">
          <span className={value ? "text-white" : "text-mute"}>
            {value ?? placeholder}
          </span>
          <span className="flex items-center gap-2 text-mute text-xs">
            {suffix}
            {type === "select" && "⌄"}
            {type === "date" && <Icon name="calendar" />}
          </span>
        </div>
      )}
    </label>
  )
}

export function Tabs({
  items,
  value,
  onChange,
}: {
  items: string[]
  value: string
  onChange: (v: string) => void
}) {
  return (
    <div className="inline-flex p-1 rounded-full bg-white/6 gap-1">
      {items.map((t) => (
        <button
          key={t}
          onClick={() => onChange(t)}
          className={`h-9 px-4 rounded-full text-sm font-semibold transition-colors ${
            t === value ? "bg-white text-abyss" : "text-secondary hover:text-white"
          }`}
        >
          {t}
        </button>
      ))}
    </div>
  )
}

export function Badge({
  children,
  strong,
}: {
  children: ReactNode
  strong?: boolean
}) {
  const text = String(children).toLowerCase()
  const positive =
    strong || /lunas|on track|sudah|tercapai|terbit|baru/.test(text)
  const danger = /belum|tinggi|melebihi|draf/.test(text)
  const warning = /sedang/.test(text)
  const tone = danger
    ? "bg-coral-soft text-coral"
    : warning
      ? "bg-amber-soft text-amber"
      : positive
        ? "bg-aqua-soft text-aqua"
        : "bg-aqua-soft text-aqua"
  return (
    <span
      className={`inline-flex items-center h-7 px-3 rounded-full text-xs font-bold ${tone}`}
    >
      {children}
    </span>
  )
}

export function Progress({ value, max = 100 }: { value: number; max?: number }) {
  const filled = Math.round(Math.min(1, value / max) * 20)
  return (
    <div
      className="grid grid-cols-[repeat(20,minmax(0,1fr))] gap-1"
      aria-label={`${Math.round((value / max) * 100)}%`}
    >
      {Array.from({ length: 20 }, (_, i) => (
        <span
          key={i}
          className={`h-3 rounded-sm ${i < filled ? "bg-gradient-to-r from-aqua to-lime" : "bg-white/10"}`}
        />
      ))}
    </div>
  )
}

export function Gauge({
  value,
  label = "kWh",
  compact = false,
}: {
  value: string
  label?: string
  compact?: boolean
}) {
  const ticks = Array.from({ length: 19 }, (_, i) => i)
  return (
    <div className={`relative ${compact ? "size-24" : "w-64 h-44"}`}>
      <svg viewBox="0 0 260 180" className="size-full" aria-label={`${value} ${label}`}>
        <defs>
          <linearGradient id="gaugeGradient" x1="30" y1="0" x2="230" y2="0">
            <stop offset="0%" stopColor="#25C6AC" />
            <stop offset="100%" stopColor="#C9F24A" />
          </linearGradient>
        </defs>
        {ticks.map((tick) => {
          const angle = -210 + tick * (240 / 18)
          const radians = (angle * Math.PI) / 180
          const x1 = 130 + Math.cos(radians) * 112
          const y1 = 136 + Math.sin(radians) * 112
          const x2 = 130 + Math.cos(radians) * 120
          const y2 = 136 + Math.sin(radians) * 120
          return (
            <line
              key={tick}
              x1={x1}
              y1={y1}
              x2={x2}
              y2={y2}
              stroke="rgba(255,255,255,.24)"
              strokeWidth="2"
              strokeLinecap="round"
            />
          )
        })}
        <path
          d="M37 190 A105 105 0 1 1 223 190"
          pathLength="100"
          fill="none"
          stroke="rgba(255,255,255,.1)"
          strokeWidth="14"
          strokeLinecap="round"
        />
        <path
          d="M37 190 A105 105 0 1 1 223 190"
          pathLength="100"
          fill="none"
          stroke="url(#gaugeGradient)"
          strokeWidth="14"
          strokeLinecap="round"
          strokeDasharray="64 100"
        />
      </svg>
      <div className="absolute inset-x-0 bottom-3 text-center">
        <span className={`${compact ? "text-lg" : "text-[42px]"} font-semibold tabular-nums`}>
          {value}
        </span>
        <span className={`${compact ? "text-[10px]" : "text-base"} text-secondary ml-1`}>
          {label}
        </span>
      </div>
    </div>
  )
}

      {["01 Okt", "03 Okt", "05 Okt", "07 Okt", "09 Okt"].map((x, i) => (
        <text key={x} x={54 + i * 171} y="257" fill="#A8BDC0" fontSize="11">
          {x}
        </text>
      ))}
    </svg>
  )
}

function BarChart({ compare = false }: { compare?: boolean }) {
  const vals = compare
    ? [176, 155, 142, 126, 108, 92, 82, 72]
    : [82, 150, 150, 196]
  return (
    <svg
      viewBox="0 0 800 260"
      className="size-full"
      preserveAspectRatio="none"
      role="img"
      aria-label="Grafik batang WattKost"
    >
      {[48, 96, 144, 192, 240].map((y) => (
        <line
          key={y}
          x1="45"
          y1={y}
          x2="765"
          y2={y}
          stroke="rgba(255,255,255,.08)"
          strokeDasharray="5 7"
        />
      ))}
      {vals.map((v, i) => (
        <rect
          key={i}
          x={compare ? 70 + i * 84 : 105 + i * 170}
          y={240 - v}
          width={compare ? 28 : 64}
          height={v}
          rx="8"
          fill={
            compare
              ? i % 2
                ? "#25C6AC"
                : "rgba(255,255,255,.24)"
              : i === 3
                ? "#25C6AC"
                : "rgba(255,255,255,.14)"
          }
        />
      ))}
      {!compare &&
        ["Juli", "Agustus", "September", "Oktober"].map((x, i) => (
          <text
            key={x}
            x={137 + i * 170}
            y="258"
            textAnchor="middle"
            fill="#A8BDC0"
            fontSize="11"
          >
            {x}
          </text>
        ))}
      {!compare && (
        <text
          x="647"
          y="34"
          textAnchor="middle"
          fill="#FFFFFF"
          fontSize="12"
          fontWeight="700"
        >
          Rp100.000
        </text>
      )}
      {compare && (
        <>
          <text x="90" y="22" fill="#A8BDC0" fontSize="12" fontWeight="600">
            September · 94,4 kWh
          </text>
          <text x="610" y="22" fill="#25C6AC" fontSize="12" fontWeight="600">
            Oktober · 86,4 kWh
          </text>
        </>
      )}
    </svg>
  )
}

export function Placeholder({
  label,
  h = "h-72",
}: {
  label: string
  h?: string
}) {
  const bar = label.includes("BATANG") || label.includes("LAPORAN")
  const compare = label.includes("PERBANDINGAN") || label.includes("LAPORAN")
  return (
    <div className={`${h} rounded-2xl bg-abyss/45 border border-line p-3 overflow-hidden`}>
      {bar ? (
        <BarChart compare={compare} />
      ) : (
        <LineChart
          spike={label.includes("LONJAKAN")}
          carbon={label.includes("EMISI")}
          projection={label.includes("PROYEKSI")}
        />
      )}
    </div>
  )
}

export function Table({
  head,
  rows,
  highlight,
  align,
}: {
  head: string[]
  rows: ReactNode[][]
  highlight?: number
  align?: ("l" | "r")[]
}) {
  return (
    <div className="border border-line rounded-2xl overflow-hidden">
      <table className="w-full text-sm">
        <thead className="bg-white/4">
          <tr>
            {head.map((h, i) => (
              <th
                key={h}
                className={`h-11 px-4 font-bold text-mute text-[11px] uppercase tracking-[0.08em] ${
                  align?.[i] === "r" ? "text-right" : "text-left"
                }`}
              >
                {h}
              </th>
            ))}
          </tr>
        </thead>
        <tbody>
          {rows.map((r, ri) => (
            <tr
              key={ri}
              className={`border-t border-line transition-colors hover:bg-aqua/6 ${
                ri === highlight ? "bg-aqua-soft outline outline-1 -outline-offset-1 outline-aqua/50" : "bg-surface"
              }`}
            >
              {r.map((c, ci) => (
                <td
                  key={ci}
                  className={`h-13 px-4 ${
                    ci > 0 ? "font-mono text-[13px]" : ""
                  } ${align?.[ci] === "r" ? "text-right" : ""}`}
                >
                  {c}
                </td>
              ))}
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  )
}

export function Modal({
  title,
  onClose,
  children,
}: {
  title: string
  onClose: () => void
  children: ReactNode
}) {
  return (
    <div
      className="absolute inset-0 z-20 bg-abyss/75 backdrop-blur-md grid place-items-center"
      onClick={onClose}
    >
      <div
        className="w-[480px] bg-raised rounded-[28px] border border-line-strong shadow-[0_24px_48px_rgba(0,0,0,.35)]"
        onClick={(e) => e.stopPropagation()}
      >
        <div className="h-16 px-6 flex items-center justify-between border-b border-line">
          <h2 className="font-display text-xl font-bold">{title}</h2>
          <button
            aria-label="Tutup"
            onClick={onClose}
            className="size-9 rounded-full bg-white/6 text-mute text-xl"
          >
            ×
          </button>
        </div>
        <div className="p-6 flex flex-col gap-4">{children}</div>
        <div className="px-6 py-4 border-t border-line flex justify-end gap-3">
          <Btn variant="secondary" onClick={onClose}>
            Batal
          </Btn>
          <Btn onClick={onClose}>Simpan</Btn>
        </div>
      </div>
    </div>
  )
}

export function Toggle({ on, onChange }: { on: boolean; onChange: () => void }) {
  return (
    <button
      onClick={onChange}
      className={`w-12 h-7 rounded-full p-1 transition-colors ${
        on ? "bg-aqua" : "bg-white/16"
      }`}
    >
      <span
        className={`block size-5 rounded-full bg-white transition-transform ${
          on ? "translate-x-5" : ""
        }`}
      />
    </button>
  )
}
