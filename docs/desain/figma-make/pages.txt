  const d = [
    [
      "AC",
      "Kurangi penggunaan AC 1 jam per hari.",
      "Matikan AC 1 jam lebih awal sebelum tidur dan gunakan kipas angin.",
      15,
      21670,
    ],
    [
      "AC",
      "Atur suhu AC ke 25°C.",
      "Setiap kenaikan 1°C menghemat sekitar 6% energi kompresor.",
      6.8,
      9824,
    ],
    [
      "Rice Cooker",
      "Matikan mode warm setelah nasi matang.",
      "Mode penghangat menyedot daya terus-menerus sepanjang hari.",
      7.2,
      10402,
    ],
    [
      "Charger HP",
      "Cabut charger setelah baterai penuh.",
      "Charger yang tetap terpasang tetap memakai daya standby.",
      0.9,
      1300,
    ],
  ] as const
  const list = d
    .map((x, i) => ({ x, i }))
    .filter(
      ({ i }) =>
        t === "Semua" || (t === "Sudah Dilakukan") === done.includes(i),
    )
  return (
    <>
      <PageTitle title="Rekomendasi Hemat">
        <Tabs
          items={["Semua", "Belum Dilakukan", "Sudah Dilakukan"]}
          value={t}
          onChange={setT}
        />
      </PageTitle>
      <div className="grid grid-cols-2 gap-6">
        {list.map(({ x: [a, j, desc, k, r], i }) => (
          <Card key={i}>
            <div className="flex justify-between items-center mb-2">
              <Badge>{a}</Badge>
              {done.includes(i) && <Badge strong>✓ Sudah dilakukan</Badge>}
            </div>
            <h2 className="text-lg font-semibold">{j}</h2>
            <p className="text-sm text-mute mt-1">{desc}</p>
            <div className="grid grid-cols-2 gap-4 my-4 p-4 rounded-lg bg-wash">
              <div>
                <p className="text-xs text-mute">Potensi hemat</p>
                <p className="font-bold">
                  {String(k).replace(".", ",")} kWh/bulan
                </p>
              </div>
              <div>
                <p className="text-xs text-mute">Potensi hemat</p>
                <p className="font-bold">{Rp(r)}/bulan</p>
              </div>
            </div>
            <Btn
              variant={done.includes(i) ? "secondary" : "primary"}
              onClick={() =>
                setDone(
                  done.includes(i) ? done.filter((n) => n !== i) : [...done, i],
                )
              }
            >
              {done.includes(i) ? "Batalkan Tanda" : "Tandai Sudah Dilakukan"}
            </Btn>
          </Card>
        ))}
      </div>
    </>
  )
}
